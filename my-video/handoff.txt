import React from 'react';
import {AbsoluteFill, interpolate, useCurrentFrame} from 'remotion';
import {Line} from './narration';

const clamp = {extrapolateLeft: 'clamp', extrapolateRight: 'clamp'} as const;
const esc = (s: string) => s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

const pieces = (text: string, terms: string[]) => {
  const list = terms.map((t) => t.trim()).filter(Boolean).sort((a, b) => b.length - a.length);
  if (list.length === 0) return [{t: text, hl: false}];
  const re = new RegExp('(' + list.map(esc).join('|') + ')', 'gi');
  return text
    .split(re)
    .filter((s) => s !== '')
    .map((s) => ({t: s, hl: list.some((l) => l.toLowerCase() === s.toLowerCase())}));
};

// White captions, key words in gold. When the creature speaks, the caption turns red and italic.
export const HighlightCaptions: React.FC<{lines: Line[]; fps: number; terms: string[]}> = ({lines, fps, terms}) => {
  const f = useCurrentFrame();
  const now = f / fps;
  const l = lines.find((x) => now >= x.start - 0.05 && now <= x.start + x.dur + 0.4);
  if (!l) return null;
  const thing = !!(l as any).thing;
  const local = (now - l.start) * fps;
  const end = l.dur * fps + 12;
  const opacity = interpolate(local, [0, 6, end - 8, end], [0, 1, 1, 0], clamp);
  return (
    <AbsoluteFill style={{justifyContent: 'flex-end', alignItems: 'center', paddingBottom: 260}}>
      <div
        style={{
          opacity,
          color: thing ? '#ff8f86' : 'white',
          fontStyle: thing ? 'italic' : 'normal',
          fontSize: 72,
          fontWeight: 800,
          fontFamily: 'Arial, sans-serif',
          textAlign: 'center',
          padding: '0 70px',
          lineHeight: 1.2,
          textShadow: '0 4px 14px rgba(0,0,0,0.9), 0 0 4px black',
        }}
      >
        {pieces(l.text, terms).map((p, i) => (
          <span
            key={i}
            style={p.hl ? {color: '#ffd54a', fontSize: 86, textShadow: '0 4px 18px rgba(0,0,0,0.95), 0 0 12px rgba(255,170,0,0.5)'} : undefined}
          >
            {p.t}
          </span>
        ))}
      </div>
    </AbsoluteFill>
  );
};

export const TitleCard: React.FC<{text: string; sub?: string; seconds: number; fps: number}> = ({text, sub, seconds, fps}) => {
  const f = useCurrentFrame();
  const end = seconds * fps;
  const opacity = interpolate(f, [0, 8, end - 10, end], [0, 1, 1, 0], clamp);
  if (!text || f > end) return null;
  return (
    <AbsoluteFill style={{justifyContent: 'flex-start', alignItems: 'center', paddingTop: 250}}>
      <div
        style={{
          opacity,
          color: 'white',
          fontSize: 112,
          fontWeight: 900,
          fontFamily: 'Arial, sans-serif',
          textAlign: 'center',
          padding: '0 60px',
          lineHeight: 1.05,
          textShadow: '0 6px 22px rgba(0,0,0,0.95), 0 0 6px black',
        }}
      >
        {text}
        {sub ? <div style={{fontSize: 60, marginTop: 18, color: '#ffd54a', letterSpacing: 6}}>{sub}</div> : null}
      </div>
    </AbsoluteFill>
  );
};

export const EndCard: React.FC<{from: number; text: string; sub: string}> = ({from, text, sub}) => {
  const f = useCurrentFrame();
  if (f < from) return null;
  const o = interpolate(f - from, [0, 14], [0, 1], clamp);
  return (
    <AbsoluteFill style={{justifyContent: 'center', alignItems: 'center', opacity: o}}>
      <div
        style={{
          color: '#ffd54a',
          fontSize: 130,
          fontWeight: 900,
          fontFamily: 'Arial, sans-serif',
          textAlign: 'center',
          textShadow: '0 6px 22px rgba(0,0,0,0.95)',
        }}
      >
        {text}
        <div style={{fontSize: 52, marginTop: 30, color: 'white', fontWeight: 700}}>{sub}</div>
      </div>
    </AbsoluteFill>
  );
};

// Orbit camera around a point. For Daniel (he faces -z): angle 0 = back view,
// 180 = front view, 90 / 270 = the two side views.
export const orbitCam = (target: V3, dist: number, angleDeg: number, height: number) => {
  const a = (angleDeg * Math.PI) / 180;
  const pos: V3 = [target[0] + Math.sin(a) * dist, height, target[2] + Math.cos(a) * dist];
  return {pos, look: target};
};

export type Segs = [number, number][];

export const smooth = (x: number) => {
  const t = Math.min(1, Math.max(0, x));
  return t * t * (3 - 2 * t);
};

// 0 = standing still, 1 = full walking speed. Eases in/out over 8 frames.
// boost = [frame, extra]: from that frame he gradually walks (1 + extra) times faster.
export const speedAt = (f: number, segs: Segs, boost?: [number, number]) => {
  let s = 0;
  for (const [a, b] of segs) s = Math.max(s, Math.min(smooth((f - a) / 8), smooth((b - f) / 8)));
  if (boost) s *= 1 + boost[1] * smooth((f - boost[0]) / 15);
  return s;
};

export const integrate = (f: number, segs: Segs, boost?: [number, number]) => {
  let dist = 0;
  let phase = 0;
  for (let i = 0; i < f; i++) {
    const s = speedAt(i, segs, boost);
    dist += 0.124 * s;
    phase += 0.34 * s;
  }
  return {dist, phase};
};

// turns: [startFrame, endFrame, targetAngle]. Each turn eases from the previous angle to its target.
export const angleKeys = (f: number, keys: [number, number, number][]) => {
  let a = 0;
  for (const [s, e, t] of keys) {
    if (f <= s) return a;
    if (f < e) return a + (t - a) * smooth((f - s) / Math.max(1, e - s));

const rgb = (hex: string) => {
  const c = parseInt(hex.slice(1), 16);
  return [(c >> 16) & 255, (c >> 8) & 255, c & 255].map((v) => (v / 255).toFixed(4));
};

const rng = (seed: number) => () => {
  seed |= 0;
  seed = (seed + 0x6d2b79f5) | 0;
  let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
};

const Sky: React.FC<{T: Theme}> = ({T}) => {
  const h = rgb(T.horizon);
  const t = rgb(T.skyTop);
  return (
    <group>
      <mesh>
        <sphereGeometry args={[180, 24, 16]} />
        <shaderMaterial
          side={THREE.BackSide}
          depthWrite={false}
          vertexShader={`
            varying vec3 vDir;
            void main() {
              vDir = normalize(position);
              gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
            }
          `}
          fragmentShader={`
            varying vec3 vDir;
            void main() {
              float k = smoothstep(0.0, 0.5, vDir.y);
              vec3 horizon = vec3(${h[0]}, ${h[1]}, ${h[2]});
              vec3 top = vec3(${t[0]}, ${t[1]}, ${t[2]});
              gl_FragColor = vec4(mix(horizon, top, k), 1.0);
            }
          `}
        />
      </mesh>
      {T.sun && (
        <>
          <mesh position={T.sun.pos}>
            <circleGeometry args={[T.sun.r, 32]} />
            <meshBasicMaterial color={T.sun.color} fog={false} />
          </mesh>
          <mesh position={[T.sun.pos[0], T.sun.pos[1], T.sun.pos[2] - 1]}>

            <meshBasicMaterial color={T.sun.glow} transparent opacity={T.sun.glowOpacity} fog={false} depthWrite={false} />

const Snow: React.FC<{frame: number; cx: number; cz: number}> = ({frame, cx, cz}) => {
  const count = 450;
  const seeds = useMemo(() => {
    const r = rng(5);
    return Array.from({length: count}, () => [r(), r(), r(), r()]);
  }, []);
  const pos = new Float32Array(count * 3);
  for (let i = 0; i < count; i++) {
    const [a, b, c, d] = seeds[i];
    const y = 9 - ((c * 9 + frame * 0.045 * (0.7 + 0.6 * d)) % 9);
    pos[i * 3] = cx + (a - 0.5) * 22 + Math.sin(frame * 0.025 + d * 6.28) * 0.4;
    pos[i * 3 + 1] = y + 0.05;
    pos[i * 3 + 2] = cz + 1 - b * 30;
  }
  return (
    <points frustumCulled={false}>
      <bufferGeometry>
        <bufferAttribute attach="attributes-position" args={[pos, 3]} />
      </bufferGeometry>
      <pointsMaterial color="#ffffff" size={0.1} sizeAttenuation transparent opacity={0.9} depthWrite={false} />
    </points>
  );
};

import json, os, sys, urllib.request, urllib.error

BASE = os.environ.get('ELEVEN_BASE', 'https://api.elevenlabs.io')


class Fail(Exception):
    pass


def key():
    k = os.environ.get('ELEVENLABS_API_KEY', '').strip()
    if not k and os.path.exists('.env'):
        for ln in open('.env'):
            if ln.strip().startswith('ELEVENLABS_API_KEY='):
                k = ln.split('=', 1)[1].strip().strip('"').strip("'")
    if not k or k == 'PASTE_YOUR_KEY_HERE':
        raise Fail('No ElevenLabs key yet. Open the file called .env and put your key after ELEVENLABS_API_KEY=')
    return k


def http(method, url, k, body=None, raw=False):
    data = json.dumps(body).encode() if body is not None else None
    req = urllib.request.Request(url, data=data, method=method,
                                 headers={'xi-api-key': k, 'Content-Type': 'application/json', 'Accept': '*/*'})
    try:
        with urllib.request.urlopen(req, timeout=180) as r:
            out = r.read()
    except urllib.error.HTTPError as e:
        raise Fail(f'ElevenLabs error {e.code}: ' + e.read().decode('utf-8', 'ignore')[:300])
    except urllib.error.URLError as e:
        raise Fail(f'Could not reach ElevenLabs: {e.reason}')
    return out if raw else json.loads(out)


def find_voice(name, k):
    want = name.lower().strip()
    names = []
    for url in (BASE + '/v1/voices', BASE + '/v2/voices?page_size=100'):
        try:
            voices = http('GET', url, k).get('voices', [])
        except Fail:
            continue
        for v in voices:
            names.append(v['name'])
            if v['name'].lower() == want:
                return v['voice_id']
        for v in voices:
            if v['name'].lower().startswith(want):
                return v['voice_id']
    raise Fail(f'Voice "{name}" not found on your account. Voices I can see: ' + ', '.join(sorted(set(names))[:30]))


def speak(text, voice_id, k, path, model='eleven_multilingual_v2', stability=0.4,
          similarity=0.8, style=0.35, speed=0.92):
    url = f'{BASE}/v1/text-to-speech/{voice_id}?output_format=mp3_44100_128'
    body = {
        'text': text,
        'model_id': model,
        'voice_settings': {'stability': stability, 'similarity_boost': similarity,
                           'style': style, 'use_speaker_boost': True, 'speed': speed},
    }
    audio = http('POST', url, k, body, raw=True)
    if len(audio) < 1000:
        raise Fail('ElevenLabs sent back almost no audio: ' + audio[:200].decode('utf-8', 'ignore'))
    open(path, 'wb').write(audio)
    return len(audio) * 8 / 128000.0

    weights = [len(t) + 2 for t in tokens]

    if CUT - 0.18 <= t < CUT + 0.15:

    else:
        env = 0.35
    chug = 0.7 + 0.3 * math.sin(TAU * 11 * tt)
    pitch = 1.0 + 0.25 * (env if tt < 2.4 else 0.0)
    v = (math.sin(TAU * 47 * pitch * tt) + 0.6 * math.sin(TAU * 94 * pitch * tt)) * chug + lp * 9
    eng.append(v * env)
if eng:
    sc = 0.11 / rms(eng)
    for q in range(len(eng)):
        eng[q] *= sc * (max(0.0, min(1.0, (TOTAL - (CUT + q / SR)) / 0.6)))
    add(mix, CUT + 0.2, eng, 1.0)

# --- heartbeat, speeding up ---
tb = CUT + 2.5
while tb < TOTAL - 0.5:
    prog = (tb - (CUT + 2.5)) / max(1.0, TOTAL - CUT - 3.0)
    bpm = 60 + 40 * min(1.0, prog)
    amp = 0.22 + 0.3 * min(1.0, prog)
    for off, a2 in ((0.0, 1.0), (0.23, 0.65)):
        n = int(0.28 * SR)
        beat = [math.sin(TAU * 52 * q / SR) * math.exp(-q / SR * 20) for q in range(n)]
        add(mix, tb + off, beat, amp * a2)
    tb += 60.0 / bpm

# --- master ---
peak = max(abs(x) for x in mix) or 1.0
scale = 0.9 / peak
pcm = array.array('h', [int(max(-1.0, min(1.0, x * scale)) * 32000) for x in mix])
with wave.open(f'public/audio/{name}/sfx.wav', 'wb') as w:
    w.setnchannels(1)
    w.setsampwidth(2)
    w.setframerate(SR)
    w.writeframes(pcm.tobytes())
print(f'sfx.wav written ({TOTAL:.1f}s)')

import os, sys

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import eleven

NAMES = ['George', 'Brian', 'Daniel', 'Callum', 'Harry', 'Will', 'Eric', 'Bill', 'Clyde', 'Roger']
TEXT = "Nobody knew my secret. Everyone in town knew I was a quiet man."

try:
    k = eleven.key()
    os.makedirs('out', exist_ok=True)
    for n in NAMES:
        try:
            vid = eleven.find_voice(n, k)
        except eleven.Fail:
            print('skipped (not on your account):', n)
            continue
        eleven.speak(TEXT, vid, k, f'out/voice_{n}.mp3')
        print('made', n)
except eleven.Fail as e:
    sys.exit('ERROR: ' + str(e))
